# Step 1.3 | Installs all runtime packages the backend needs
# 
# express         → HTTP server + routing
# mongoose        → MongoDB ODM, schema validation
# dotenv          → loads .env into process.env
# bcryptjs        → hashes passwords before saving to DB (never store plain text)
# jsonwebtoken    → signs and verifies JWT tokens for auth
# cookie-parser   → parses cookies from incoming requests (we store JWT in a cookie)
# cors            → allows frontend (different port) to call backend without CORS errors
# cloudinary      → SDK to upload images to Cloudinary CDN
# socket.io       → real-time bidirectional events between server and clients
# @arcjet/node    → rate limiting middleware (blocks brute force / spam)
# resend          → sends transactional emails (welcome, OTP, etc.)

npm install express mongoose dotenv bcryptjs jsonwebtoken cookie-parser cors cloudinary socket.io @arcjet/node resend

# Step 1.4 | Installs developer-only tools (not bundled into production)
#
# nodemon → watches files and auto-restarts the server on save
#           Interview line: "nodemon eliminates the manual restart loop during development"

npm install -D nodemon
